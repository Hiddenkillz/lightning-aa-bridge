# aasdk's exported targets were generated by a native build inside the Pi
# image, so their INTERFACE_LINK_LIBRARIES contain absolute paths like
# /usr/lib/aarch64-linux-gnu/libusb-1.0.so. When cross-compiling against the
# image as a sysroot, those paths must be prefixed with the sysroot or the
# linker (and make, as a prerequisite) looks for them on the host.
if(CMAKE_SYSROOT)
  foreach(_tgt AASDK::aasdk AAPProto::aap_protobuf)
    if(TARGET ${_tgt})
      get_target_property(_libs ${_tgt} INTERFACE_LINK_LIBRARIES)
      if(_libs)
        set(_fixed "")
        foreach(_l ${_libs})
          if(_l MATCHES "^/usr/lib/")
            list(APPEND _fixed "${CMAKE_SYSROOT}${_l}")
          else()
            list(APPEND _fixed "${_l}")
          endif()
        endforeach()
        set_target_properties(${_tgt} PROPERTIES INTERFACE_LINK_LIBRARIES "${_fixed}")
      endif()
    endif()
  endforeach()
endif()
